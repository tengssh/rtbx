* Overview
NetCDF (Network Common Data Form) is a set of freely-distributed open source libraries used to create portable, self-describing, and appendable data formats.
It is one of the popular scientific data standards that support several programming languages (e.g., C/C++, Fortran, and Python).

NetCDF version 4.0 (NetCDF-4), in particular, uses HDF5 as its underlying format.
As a result, NetCDF-4 files can be read by HDF5 and support parallel I/O (using HDF5 library), but they cannot be read by versions of netCDF older than 4.0.

** [[https://docs.unidata.ucar.edu/netcdf-c/current/netcdf_data_model.html][The NetCDF data model]]
A NetCDF file may contain the following components (syntax in C):
- File [/ Group (for >=v4.0)]
  - Subgroups (for >=v4.0)
  - User-defined types (for >=v4.0)
  - Variables: Data containers
    - name
    - type (DataType)
    - shape (Dimension)
  - Attributes: Metadata
    - name
    - type (DataType)
    - values
  - Dimensions: Shape of variables
    - name
    - length = fixed or =NC_UNLIMITED=
- DataType
  - PrimitiveType
    - =NC_CHAR/BYTE/SHORT/INT/FLOAT/DOUBLE=
    - =NC_INT64/UBYTE/USHORT/UINT/UINT64/STRING= (for >=v4.0)
  - User-defined types (for >=4.0)
    - Enum, Opaque, Compound, VariableLength

** NetCDF workflow
The following is a typical workflow for creating a netCDF file (syntax in C):
1. Create/Open File (=nc_create= / =nc_open=)
2. Define metadata (=nc_def_dim=, =nc_def_var=, =nc_put_att*=) -> (=nc_enddef=)
3. Write/Read data (=nc_put_var*= / =nc_inq_grp_ncid=, =nc_inq_varid=, =nc_get_att*=, =nc_get_var*=)
4. Close File (=nc_close=)

- Define data

- Create file
  - Creation mode flags
    - =NC_CLOBBER=: Overwrite if the file exists
    - =NC_NOCLOBBER=: Create a new file only if the file does not exist
    - File formats
      - Default: CDF-1
      - =NC_64BIT_OFFSET=: CDF-2
      - =NC_64BIT_DATA=: CDF-5
      - =NC_NETCDF4=: NetCDF-4/HDF5
  - Open mode flags
    - =NC_NOWRITE=: Read-only
    - =NC_WRITE=: Read-write access
#+begin_src c :tangle create_netcdf4.c
    const char file_name[] = "trajectories.nc";
    int status = NC_NOERR;
    int ncid;
    nc_create(file_name, NC_CLOBBER | NC_NETCDF4, &ncid);
#+end_src

- Define metadata
#+begin_src c :tangle create_netcdf4.c
    // Group
    const char group_name[] = "Trajectories";
    int group_id;
    nc_def_grp(ncid, group_name, &group_id);
    nc_put_att_text(group_id, NC_GLOBAL, "Boundary condition", strlen("PBC"), "PBC");
    nc_put_att_int(group_id, NC_GLOBAL, "Number of samples", NC_INT, 1, &n_samples);

    // Dimension
    int sample_dimid;
    nc_def_dim(group_id, "sample", n_samples, &sample_dimid);

    int dimids[1] = {sample_dimid};

    // Variable 1
    const char ds1_name[] = "Distances";
    int ds1_id;
    nc_def_var(group_id, ds1_name, NC_INT, 1, dimids, &ds1_id);
    // Attribute 1
    nc_put_att_text(group_id, ds1_id, "Description", strlen("Distances from the origin"), "Distances from the origin");
    nc_put_att_text(group_id, ds1_id, "Unit", 1, "m");

    // Variable 2
    const char ds2_name[] = "Velocities";
    int ds2_id;
    nc_def_var(group_id, ds2_name, NC_DOUBLE, 1, dimids, &ds2_id);
    // Attribute 2
    nc_put_att_text(group_id, ds2_id, "Description", strlen("Velocities of the object"), "Velocities of the object");
    nc_put_att_text(group_id, ds2_id, "Unit", 3, "m/s");

    // End define mode
    nc_enddef(ncid);
#+end_src

*** Reading NetCDF files in C
#+begin_src c :tangle read_netcdf4.c
#include <netcdf.h>
#include <stdio.h>

int main(){
    // Open file
    int ncid;
    nc_open("trajectories.nc", NC_NOWRITE, &ncid);

    // Open group
    int group_id = ncid;
    nc_inq_grp_ncid(ncid, "Trajectories", &group_id);
    printf("Group: %s\n", group_id == ncid ? "(root)" : "Trajectories");

    // Get group attributes
    char gattr1_val[16] = {0};
    nc_get_att_text(group_id, NC_GLOBAL, "Boundary condition", gattr1_val);
    printf("Boundary condition: %s\n", gattr1_val);

    int n_samples = 0;
    nc_get_att_int(group_id, NC_GLOBAL, "Number of samples", &n_samples);
    printf("Number of samples: %d\n", n_samples);

    // Read variable 1
    int dist_varid;
    nc_inq_varid(group_id, "Distances", &dist_varid);
    char ds1_attr1_val[128] = {0};
    nc_get_att_text(group_id, dist_varid, "Description", ds1_attr1_val);
    char ds1_attr2_val[16] = {0};
    nc_get_att_text(group_id, dist_varid, "Unit", ds1_attr2_val);
    int distances[10];
    nc_get_var_int(group_id, dist_varid, distances);

    printf("\nDataset: Distances\n  Description: %s\n  Unit: %s\n",
           ds1_attr1_val, ds1_attr2_val);
    for (int i = 0; i < n_samples; i++) {
        printf("%d ", distances[i]);
    }
    printf("\n");

    // Read variable 2
    int vel_varid;
    nc_inq_varid(group_id, "Velocities", &vel_varid);
    char ds2_attr1_val[128] = {0};
    nc_get_att_text(group_id, vel_varid, "Description", ds2_attr1_val);
    char ds2_attr2_val[16] = {0};
    nc_get_att_text(group_id, vel_varid, "Unit", ds2_attr2_val);
    double velocities[10];
    nc_get_var_double(group_id, vel_varid, velocities);

    printf("\nDataset: Velocities\n  Description: %s\n  Unit: %s\n",
           ds2_attr1_val, ds2_attr2_val);
    for (int i = 0; i < n_samples; i++) {
        printf("%.1f ", velocities[i]);
    }
    printf("\n");

    // Close file
    nc_close(ncid);
    return 0;
}
