* Overview
HDF5 (Hierarchical Data Format, Version 5) is an open-source, portable, self-describing data format.
It is widely used in scientific communities as it can store large amounts of data, supports parallel I/O, and offers APIs in many languages (e.g., C/C++, Fortran, and Python).

** HDF5 objects
The HDF5 format contain the following data objects:
- Files: Containers for storing HDF5 data
- Groups: File-system-like folders for data organization
- Datasets: Actual data
- Datatypes: Definitions of data format (e.g., integer, float, string)
- Dataspaces: Dimensionality and shape of the data
- Property List: Customizable settings for storage behaviors (e.g., compression, chunking, and file access tuning)
- Attributes: Metadata for contextual descriptions of data objects (except for attributes)

** HDF5 workflow
In principle, the workflow to access an HDF5 file follows a hierarchical order, for example (syntax in C):
1. Create/Open File (=H5Fcreate= / =H5Fopen=)
  2. Create/Open Group (=H5Gcreate= / =H5Gopen=)
    3. Use/Copy Datatype (e.g., =H5T_NATIVE_INT=, =H5Tcopy(H5T_C_S1)=)
    4. Create/Open Dataspace (e.g., =H5Screate_simple=)
    5. Create/Open Property List (=H5Pcreate=: optional, or use =H5P_DEFAULT=)
      6. Create/Open Dataset (=H5Dcreate=: using Datatype, Dataspace, and Property IDs)
        7. Create/Open Attribute (=H5Acreate=: attached to Dataset or Group)
          8. Write/Read data
        9. Close Attribute (=H5Aclose=)
      10. Close Dataset (=H5Dclose=)
    11. Close Property (=H5Pclose=)
    12. Close Dataspace (=H5Sclose=)
    13. Close Datatype (=H5Tclose=: for copied ones)
  14. Close Group (=H5Gclose=)
15. Close File (=H5Fclose=)

*** Reading HDF5 files in C
#+begin_src c :tangle read_hdf5.c
#include <hdf5.h>
#include <stdio.h>

int main(){
    // Open file
    hid_t file_id = H5Fopen("trajectories.hdf5", H5F_ACC_RDONLY, H5P_DEFAULT);

    // Open group
    hid_t group_id = H5Gopen(file_id, "Trajectories", H5P_DEFAULT);

    // Open group attributes
    hid_t gattr1_id = H5Aopen(group_id, "Boundary condition", H5P_DEFAULT);
    hid_t type_str = H5Aget_type(gattr1_id);
    char gattr1_val[10];
    H5Aread(gattr1_id, type_str, gattr1_val);
    H5Aclose(gattr1_id);
    H5Tclose(type_str);
    printf("Boundary condition: %s\n", gattr1_val);

    hid_t gattr2_id = H5Aopen(group_id, "Number of samples", H5P_DEFAULT);
    int n_samples = 0;
    H5Aread(gattr2_id, H5T_NATIVE_INT, &n_samples);
    H5Aclose(gattr2_id);
    printf("Number of samples: %d\n", n_samples);

    // Open dataset 1
    hid_t ds1_id = H5Dopen(group_id, "Distances", H5P_DEFAULT);
    int distances[10];
    H5Dread(ds1_id, H5T_NATIVE_INT, H5S_ALL, H5S_ALL, H5P_DEFAULT, distances);

    hid_t ds1_attr1 = H5Aopen(ds1_id, "Description", H5P_DEFAULT);
    hid_t ds1_type1 = H5Aget_type(ds1_attr1);
    char ds1_attr1_val[128] = {0};
    H5Aread(ds1_attr1, ds1_type1, ds1_attr1_val);
    H5Aclose(ds1_attr1);
    H5Tclose(ds1_type1);

    hid_t ds1_attr2 = H5Aopen(ds1_id, "Unit", H5P_DEFAULT);
    hid_t ds1_type2 = H5Aget_type(ds1_attr2);
    char ds1_attr2_val[10] = {0};
    H5Aread(ds1_attr2, ds1_type2, ds1_attr2_val);
    H5Aclose(ds1_attr2);
    H5Tclose(ds1_type2);

    printf("\nDataset: Distances\n  Description: %s\n  Unit: %s\n", ds1_attr1_val, ds1_attr2_val);
    for(int i = 0; i < n_samples; i++) {
        printf("%d ", distances[i]);
    }
    printf("\n");
    H5Dclose(ds1_id);
    
    // Open dataset 2
    hid_t ds2_id = H5Dopen(group_id, "Velocities", H5P_DEFAULT);
    double velocities[10];
    H5Dread(ds2_id, H5T_NATIVE_DOUBLE, H5S_ALL, H5S_ALL, H5P_DEFAULT, velocities);

    hid_t ds2_attr1 = H5Aopen(ds2_id, "Description", H5P_DEFAULT);
    hid_t ds2_type1 = H5Aget_type(ds2_attr1);
    char ds2_attr1_val[128] = {0};
    H5Aread(ds2_attr1, ds2_type1, ds2_attr1_val);
    H5Aclose(ds2_attr1);
    H5Tclose(ds2_type1);

    hid_t ds2_attr2 = H5Aopen(ds2_id, "Unit", H5P_DEFAULT);
    hid_t ds2_type2 = H5Aget_type(ds2_attr2);
    char ds2_attr2_val[10] = {0};
    H5Aread(ds2_attr2, ds2_type2, ds2_attr2_val);
    H5Aclose(ds2_attr2);
    H5Tclose(ds2_type2);

    printf("\nDataset: Velocities\n  Description: %s\n  Unit: %s\n", ds2_attr1_val, ds2_attr2_val);
    for(int i = 0; i < n_samples; i++) {
        printf("%.1f ", velocities[i]);
    }
    printf("\n");
    H5Dclose(ds2_id);

    // Close everything
    H5Gclose(group_id);
    herr_t status = H5Fclose(file_id);
    return status;
}
